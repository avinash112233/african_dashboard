def calculate_aqi(pm25):
    # Handle missing values
    if pd.isna(pm25):
        return np.nan  # Return NaN (blank) if PM2.5 value is missing

    # AQI 0–50: "Good"
    # Air quality is satisfactory; little or no risk
    # NOTE: You are using 0–9.0 µg/m³ (stricter than EPA standard 0–12.0)
    elif pm25 <= 9.0:
        return linear_conversion(pm25, 0, 9.0, 0, 50)

    # AQI 51–100: "Moderate"
    # Acceptable air quality; some pollutants may be a concern for sensitive individuals
    elif pm25 <= 35.4:
        return linear_conversion(pm25, 9.1, 35.4, 51, 100)

    # AQI 101–150: "Unhealthy for Sensitive Groups (USG)"
    # Sensitive groups (e.g., children, elderly, respiratory patients) may experience health effects
    elif pm25 <= 55.4:
        return linear_conversion(pm25, 35.5, 55.4, 101, 150)

    # AQI 151–200: "Unhealthy"
    # Everyone may begin to experience health effects; sensitive groups more serious
    elif pm25 <= 125.4:
        return linear_conversion(pm25, 55.5, 125.4, 151, 200)

    # AQI 201–300: "Very Unhealthy"
    # Health alert; increased likelihood of adverse effects for the entire population
    elif pm25 <= 225.4:
        return linear_conversion(pm25, 125.5, 225.4, 201, 300)

    # AQI 301–500: "Hazardous"
    # Emergency conditions; serious health effects for everyone
    elif pm25 <= 500.0:
        return linear_conversion(pm25, 225.5, 500.0, 301, 500)

    # Beyond AQI scale
    else:
        return "Out of Range"
